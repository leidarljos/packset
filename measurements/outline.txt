import json, statistics, subprocess
files = subprocess.check_output(["git", "ls-files"], text=True).splitlines()
lens = [len(p) for p in files]
print(f"tracked {len(files)}")
print(f"path_bytes mean {statistics.mean(lens):.1f} median {statistics.median(lens):.0f}")

def synth(n):
    return [f"{files[i % len(files)]}-extra-{i}" for i in range(n)]

def dump(n, listed_cap, too_large):
    paths = synth(n)
    if n > too_large:
        shown, status = [], "too-large"
    else:
        shown, status = paths[:listed_cap], "synced"
    raw = json.dumps({"status": status, "count": n, "listed": len(shown), "files": shown})
    print(f"n={n} status={status} listed={len(shown)} json_bytes={len(raw)}")

dump(500, 500, 5000)
